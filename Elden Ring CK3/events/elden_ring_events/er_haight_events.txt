		}
	}

#Kenneth expands his realm, either by getting mistwood or gaining enough land
er_haight.0002 = {
	type = character_event
	title = er_haight.0002.t
	desc = er_haight.0002.desc
	theme = realm
	left_portrait = {
		character = root
		animation = personality_rational
	}

	immediate = {
		add_prestige = 1500
		dynasty = { add_dynasty_prestige = 150 }
	}
	
	option = {
		name = er_haight.0002.a
		remove_trait = content
		add_trait = ambitious
		add_pressed_claim = title:k_stormveil
		
		#effect = {
		#	set_relation_rival = title:k_stormveil.holder
		#}
		
		hidden_effect = {
			if = {
				limit = {
					title:k_stormveil = { is_title_created = yes }
				}
				title:k_stormveil.holder = {
					set_relation_rival = {
						target = character:haight
						reason = rival_claimed_my_title
						involved_character = root
					}
					trigger_event = { id = er_stormveil.0001 }
				}
				reverse_add_opinion = {
					modifier = angry_opinion
					target = title:k_stormveil.holder
					opinion = -100
				}
			}
		}
		
		ai_chance = {
			base = 30
		}
	}
	
	option = {
		name = er_haight.0002.b
		
		stress_impact = {
			content = medium_stress_impact_loss
		}
		
		ai_chance = {
			base = 70
		}
	}
}
