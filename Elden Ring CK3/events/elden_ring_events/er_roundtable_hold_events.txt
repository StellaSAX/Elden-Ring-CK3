#get to the table
er_roundtable_hold.0001 = {
	type = character_event
	title = er_roundtable_hold.0001.t
	window = visit_settlement_window
	desc = er_roundtable_hold.0001.desc
	theme = landless_adventurer

	override_background = { reference = roundtable_main_hall }
	
	center_portrait = {
		character = root
		triggered_animation = {
			trigger = { 1010_has_irrational_traits = yes }
			animation = personality_irrational
		}
		triggered_animation = {
			trigger = { 1010_has_standoffish_traits = yes }
			animation = personality_coward
		}
		triggered_animation = {
			trigger = { 1010_has_uninterested_traits = yes }
			animation = personality_callous
		}
		animation = personality_content
	}
	
	#talk to leader
	option = {
		name = er_roundtable_hold.0001.a
		trigger = {
			custom_tooltip = {
				text = er_roundtable_hold.0002.cooldown.tt
				NAND = {
					has_variable = hired_tarnished_in_roundtable
					has_variable = increse_skill_in_roundtable
				}
			}
		}
		
		#Actual transition event
		custom_tooltip = er_roundtable_hold.0001.a.tt
		trigger_event = er_roundtable_hold.0002
		ai_chance = { base = 0 }
	}
	#buy artifact
	option = {
		name = er_roundtable_hold.0001.b
		
		#Actual transition event
		custom_tooltip = er_roundtable_hold.0001.b.tt
		trigger_event = er_roundtable_hold.0003
		ai_chance = { base = 0 }
	}
	#change faith
	option = {
		name = er_roundtable_hold.0001.c
		
		#Actual transition event
		custom_tooltip = er_roundtable_hold.0001.c.tt
		trigger_event = er_roundtable_hold.0004
		ai_chance = { base = 0 }
	}
	#get out
	option = {
		name = er_roundtable_hold.1000.d
		custom_tooltip = er_roundtable_hold.1000.d.tt
		#For loc reasons
		1000_remember_settlement_effect = yes
		1000_clean_up_effect = yes
		ai_chance = { base = 50 }
	}
}

				}

					set_character_faith = faith:faith_perfect
				}
				
			}
		
		}

		add_courtier = scope:random_tarnished_to_hire
		ai_chance = {
			base = 50
			modifier = {
				short_term_gold < scope:second_recruit_cost
				short_term_gold < laamp_ai_spending_pants_value
				any_courtier = { count <= 2 }
				factor = 0
			}
		}
	}

				text = er_roundtable_hold.0002.cooldown.norune
