imprison_interaction = {
	interface_priority = 120
	common_interaction = yes
	icon = prison

	category = interaction_category_hostile
	highlighted_reason = {
		triggered_desc = {
			trigger = {
				scope:puppet_or_actor = {
					has_title = title:e_minister_of_justice
					NOT = { has_imprisonment_reason = scope:recipient }
				}
			}
			desc = {
				first_valid = {
					triggered_desc = {
						trigger = { NOT = { scope:actor = scope:puppet_or_actor } }
						desc = HIGHLIGHTED_MINISTER_IMPRISONMENT_REASON_puppeteer
					}
					desc = HIGHLIGHTED_MINISTER_IMPRISONMENT_REASON
				}
			}
		}
		triggered_desc = {
			trigger = {
				scope:puppet_or_actor = {
					has_imprisonment_reason = scope:recipient
				}
			}
			desc = {
				first_valid = {
					triggered_desc = {
						trigger = { NOT = { scope:actor = scope:puppet_or_actor } }
						desc = HIGHLIGHTED_HAS_IMPRISONMENT_REASON_puppeteer
					}
					desc = HIGHLIGHTED_HAS_IMPRISONMENT_REASON
				}
			}
		}
	}
	ai_maybe = yes
	ai_min_reply_days = 4
	ai_max_reply_days = 9
	can_send_despite_rejection = yes
	popup_on_receive = yes
	pause_on_receive = yes

	desc = imprison_interaction_desc

	greeting = negative
	notification_text = REQUEST_ARREST_NOTIFICATION_TEXT
	pre_answer_maybe_key = ANSWER_MIGHT_SUCCEED
	pre_answer_no_key = ANSWER_CANT_SUCCEED
	pre_answer_yes_key = ANSWER_WILL_SUCCEED
	pre_answer_maybe_breakdown_key = ANSWER_SUM_CHANCE

	on_decline_summary = {
		triggered_desc = {
			trigger = {
				scope:recipient = {
					NOR = {
						AND = {
							has_government = landless_adventurer_government
							reverse_has_opinion_modifier = {
								target = scope:puppet_or_actor
								modifier = eviction_ignored_opinion
							}
						}
						scope:puppet_or_actor = {
							has_title = title:e_minister_of_justice
						}
						scope:recipient = {
							is_landed = no
							OR = {
								exists = clerical_elector_title
								any_held_title = {
									OR = {
										is_noble_family_title = yes
										has_clerical_region = yes
									}
								}
							}
						}
					}
				}
			}
			desc = imprison_decline_summary
		}
	}

	is_shown = {
		scope:puppet_or_actor ?= {
			this != scope:recipient
			OR = {
				AND = {
					has_title = title:e_minister_of_justice
					scope:recipient = {
						tgp_is_any_minister = no
						liege ?= {
							basic_allowed_to_imprison_character_trigger = {
								CHARACTER = scope:recipient
							}
						}
					}
				}
				basic_allowed_to_imprison_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
			trigger_if = {
				limit = {
					is_ai = yes
					NOT = { puppeteer ?= { is_ai = no } }
				}
				trigger_if = {
					limit = { has_realm_law_flag = imprisonment_toggle_enable }
					has_realm_law_flag = imprisonment_toggle_on
				}
				is_imprisoned = no
			}
		}
		scope:recipient = {
			NOT = { cannot_be_imprisoned = yes }
		}
		# Scope:recipient can't be visiting elsewhere.
		trigger_if = {
			limit = {
				scope:recipient = { is_courtier_of = scope:puppet_or_actor }
			}
			NOT = {
				scope:recipient.host = {
					any_foreign_court_guest = { this = scope:recipient }
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:puppet_or_actor = {
			trigger_if = {
				limit = {
					has_title = title:e_minister_of_justice
					scope:recipient = {
						tgp_is_any_minister = no
					}
					NOT = {
						allowed_to_imprison_character_trigger = { CHARACTER = scope:recipient }
					}
				}
				top_liege = {
					allowed_to_imprison_character_trigger = { CHARACTER = scope:recipient }
				}
			}
			trigger_else = {
				allowed_to_imprison_character_trigger = { CHARACTER = scope:recipient }
			}
			trigger_if = {
				limit = {
					government_has_flag = government_is_nomadic
				}
				NOT = { has_realm_law = nomadic_authority_1 }
			}
			trigger_if = {
				limit = {
					has_title = title:e_minister_of_justice
				}
				scope:recipient = {
					NOT = {
						is_close_or_extended_family_of = scope:puppet_or_actor.top_liege
					}
				}
			}
		}
		imprison_neutral_is_valid_showing_failures_only_triggers_trigger = yes
		trigger_if = {
			limit = { scope:actor = scope:puppet_or_actor }
			custom_tooltip = {
				text = cannot_take_overt_hostile_actions_against_diarch.tt
				NOT = { scope:recipient ?= scope:puppet_or_actor.diarch }
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = cannot_take_overt_hostile_actions_against_diarch.tt_puppeteer
				NOT = { scope:recipient ?= scope:puppet_or_actor.diarch }
			}
		}
		trigger_if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				prestige >= liege_diarchy_surcharge_interaction_imprison_interaction_value
			}
		}
		trigger_if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				treasury >= {
					value = 0
					if = {
						limit = {
							scope:puppet_or_actor = {
								has_title = title:e_minister_of_justice
							}
							scope:recipient = {
								NOT = { is_courtier_of = scope:puppet_or_actor }
							}
						}
						add = scope:puppet_or_actor.major_treasury_value
					}
				}
			}
		}
	}

	can_send = {
		scope:puppet_or_actor = {
			custom_description = {
				text = "character_interactions_hostile_actions_disabled_delay"
				NOT = { has_character_flag = flag_hostile_actions_disabled_delay }
			}
			trigger_if = {
				limit = {
					scope:actor = { is_ai = no }
				}
				scope:recipient = {
					trigger_if = {
						limit = {
							has_government = landless_adventurer_government
							reverse_has_opinion_modifier = {
								target = scope:puppet_or_actor
								modifier = eviction_ignored_opinion
							}
						}
						is_travelling = no
					}
				}
			}
			trigger_else = {
				OR = {
					has_imprisonment_reason = scope:recipient
					any_character_struggle = {
						involvement = involved
					}
				}
			}
		}
	}

	is_highlighted = {
		scope:recipient = {
			is_imprisoned = no
		}
		scope:puppet_or_actor = {
			OR = {
				has_imprisonment_reason = scope:recipient
				has_title = title:e_minister_of_justice
			}
		}
	}

	#Use hook
	send_option = {
		is_valid = {
			exists = scope:recipient
			scope:puppet_or_actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = SCHEME_HOOK
	}
	should_use_extra_icon = {
		scope:puppet_or_actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	send_options_exclusive = no

	cost = {
		# Usually, this action is free, but during certain diarchies, we want it to cost.
		prestige = {
			value = liege_diarchy_surcharge_interaction_imprison_interaction_value
			if = {
				limit = { scope:is_puppet_action ?= yes }
				multiply = 0
			}
		}
		treasury = {
			if = {
				limit = {
					scope:puppet_or_actor = {
						has_title = title:e_minister_of_justice
					}
					scope:recipient = {
						NOT = { is_courtier_of = scope:puppet_or_actor }
					}
				}
				value = {
					value = 0
					desc = BASE
				}
				add = {
					value = scope:puppet_or_actor.major_treasury_value
					desc = IMPRISON_MINISTER_OF_JUSTICE_TREASURY_COST
				}
			}
			if = {
				limit = { scope:is_puppet_action ?= yes }
				multiply = 0
			}
		}
	}

	on_send = {
		scope:puppet_or_actor = {
			add_character_flag = {
				flag = flag_hostile_actions_disabled_delay
				days = 10
			}
		}
	}

	on_auto_accept = {
		# the rest of the necessary logic is covered by auto_accept also triggering on_accept below
		scope:recipient = {
			trigger_event = char_interaction.0233
		}
	}

	on_accept = {
		if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				if = {
					limit = { liege_diarchy_surcharge_interaction_imprison_interaction_value > 0 }
					add_prestige = { value = { add = liege_diarchy_surcharge_interaction_imprison_interaction_value max = prestige multiply = -1 } }
				}
				if = {
					limit = {
						has_title = title:e_minister_of_justice
						scope:recipient = {
							NOT = { is_courtier_of = scope:puppet_or_actor }
						}
					}
					remove_treasury = scope:puppet_or_actor.major_treasury_value
				}
			}
		}
		scope:puppet_or_actor = {
			if = {
				limit = {
					has_title = title:e_minister_of_justice
				}
				if = {
					limit = {
						NOR = {
							has_imprisonment_reason = scope:recipient
							scope:recipient.liege = {
								has_imprisonment_reason = scope:recipient
							}
							scope:hook = yes
						}
						any_character_situation = {
							situation_type = dynastic_cycle
							situation_top_has_catalyst = catalyst_minister_imprison
						}
					}
					save_scope_value_as = {
						name = use_strife_not_tyranny
						value = yes
					}
					situation:dynastic_cycle = {
						trigger_situation_catalyst = {
							catalyst = catalyst_minister_imprison
							character = scope:recipient
						}
					}
				}
				if = {
					limit = {
						government_allows = merit
					}
					change_merit = {
						value = minor_merit_gain
						multiply = scope:recipient.highest_held_title_tier
					}
				}
				if = {
					limit = {
						government_has_flag = government_has_influence
					}
					change_influence = {
						value = minor_influence_gain
						multiply = scope:recipient.highest_held_title_tier
					}
				}
				scope:recipient.liege = {
					if = {
						limit = { has_imprisonment_reason = scope:recipient }
						consume_imprisonment_reasons = scope:recipient
					}
				}
			}
		}
		if = {
			limit = { scope:hook = yes }
			scope:puppet_or_actor = { use_hook = scope:recipient }
		}
		illegal_imprisonment_dread_effect = yes
		imprison_court_chaplain_consequences_effect = yes
		imprison_HoF_consequences_effect = yes
		scope:recipient = {
			if = {
				limit = {
					is_a_faction_member = yes
					liege = scope:puppet_or_actor
				}
				joined_faction = { save_scope_as = retaliating_faction }
			}
		}
		imprison_character_effect = {
			TARGET = scope:recipient
			IMPRISONER = scope:puppet_or_actor
		}
		# If they recently revoked your title, or asked for your excommunication, they become your rival
		imprison_attempt_escalates_rivalry_effect = yes
		scope:puppet_or_actor = {
			trigger_event = char_interaction.0231
			if = {
				limit = {
					has_imprisonment_reason = scope:recipient
				}
				stress_and_fulfillment_impact = {
					compassionate = minor_stress_impact_gain
					forgiving = minor_stress_impact_gain
				}
			}
			else = {
				stress_and_fulfillment_impact = {
					just = medium_stress_impact_gain
					compassionate = medium_stress_impact_gain
				}
				if = {
					limit = { exists = scope:retaliating_faction}
					scope:retaliating_faction = {
						imprisonment_retribution_start_faction_war_if_valid = yes
					}
				}
			}
		}
		if = {
			limit = {
				scope:puppet_or_actor = { has_character_flag = flag_hostile_actions_disabled_delay }
			}
			scope:puppet_or_actor = { remove_character_flag = flag_hostile_actions_disabled_delay }
		}

		# If we're a clan this interaction affects unity
		if = { # If you don't have an imprisonment reason, and the target is landed, you lose more unity
			limit = {
				NOT = { scope:puppet_or_actor = { has_imprisonment_reason = scope:recipient } }
				scope:recipient = { is_landed_or_landless_administrative_or_theocratic = yes }
			}
			add_clan_unity_interaction_effect = {
				CHARACTER = scope:puppet_or_actor
				TARGET = scope:recipient
				VALUE = major_unity_loss
				DESC = clan_unity_imprison.desc
				REVERSE_NON_HOUSE_TARGET = no
			}
		}
		else_if = { # If you don't have an imprisonment reason, and the target is unlanded, you lose slightly less unity
			limit = {
				NOT = { scope:puppet_or_actor = { has_imprisonment_reason = scope:recipient } }
				scope:recipient = { is_landed_or_landless_administrative_or_theocratic = no }
			}
			add_clan_unity_interaction_effect = {
				CHARACTER = scope:puppet_or_actor
				TARGET = scope:recipient
				VALUE = medium_unity_loss
				DESC = clan_unity_imprison.desc
				REVERSE_NON_HOUSE_TARGET = no
			}
		}
		else = { # If you have a valid reason, you lose only a tiny bit of unity (criminal or not, they are still family)
			add_clan_unity_interaction_effect = {
				CHARACTER = scope:puppet_or_actor
				TARGET = scope:recipient
				VALUE = minor_unity_loss
				DESC = clan_unity_imprison.desc
				REVERSE_NON_HOUSE_TARGET = no
			}
		}

		# Harrying logic.
		scope:puppet_or_actor = {
			if = {
				limit = {
					character_is_valid_for_harrying_of_the_north_trigger = yes
					scope:recipient = {
						character_is_valid_for_harrying_of_the_north_trigger = yes
						is_important_or_vip_struggle_character = yes
					}
					culture != scope:recipient.culture
				}
				global_var:harrying_of_the_north = { save_scope_as = story }
				if = {
					limit = {
						character_is_valid_norman_for_harrying_of_the_north_trigger = yes
						scope:recipient = { character_is_valid_anglo_saxon_for_harrying_of_the_north_trigger = yes }
					}
					ep3_increase_pacification_effect = { AMOUNT = harrying_minor_increase_value }
				}
				else_if = {
					limit = {
						character_is_valid_anglo_saxon_for_harrying_of_the_north_trigger = yes
						scope:recipient = { character_is_valid_norman_for_harrying_of_the_north_trigger = yes }
					}
					ep3_increase_resistance_effect = { AMOUNT = harrying_minor_increase_value }
				}
			}
		}
	}

	on_decline = {
		#Close family might get rival progress
		if = {
			limit = {
				scope:recipient = {
					is_close_family_of = scope:puppet_or_actor
				}
				#Legal imprisonment
				trigger_if = {
					limit = {
						scope:puppet_or_actor = {
							has_imprisonment_reason = scope:recipient
						}
					}
					scope:recipient = {
						NOR = {
							has_trait = forgiving
							ai_honor >= low_positive_ai_value
						}
					}
				}
				#Illegal imprisonment
				trigger_else = {
					scope:recipient = {
						NOT = {
							has_trait = forgiving
						}
					}
				}
			}
			scope:recipient = {
				progress_towards_rival_effect = {
					CHARACTER = scope:puppet_or_actor
					REASON = rival_imprisoned_family
					OPINION = 0
				}
			}
		}
		if = {
			limit = {
				scope:puppet_or_actor = {
					NOT = { has_title = title:e_minister_of_justice }
				}
			}
			if = {
				limit = {
					scope:recipient = {
						has_government = landless_adventurer_government
					}
				}
				scope:recipient = {
					if = {
						limit = { current_military_strength >= ep3_interactions_events_0500_troops_count_value }
						start_war = {
							cb = ep3_laamp_apprehend_adventurer_cb
							target = scope:puppet_or_actor
						}
					}
					else = {
						# Remove them immediately.
						remove_laamp_from_exiled_county_automatically_effect = {
							PROVINCE = scope:recipient.location
							LAAMP = scope:recipient
						}
					}
				}
			}
			else = {
				scope:puppet_or_actor = {
					imprison_HoF_consequences_effect = yes
					imprison_tyranny_effect = yes
				}
				scope:recipient = {
					#If target is count or higher, start war.
					if = {
						limit = {
							is_playable_character = yes
							primary_title.tier > tier_barony
						}
						if = {
							limit = {
								is_a_faction_member = yes
								liege = scope:puppet_or_actor
								joined_faction = {
									faction_target = scope:puppet_or_actor
									NOT = { faction_is_type = populist_faction }
								}
							}
							joined_faction = {
								save_scope_as = retaliating_faction # For loc
								imprisonment_retribution_start_faction_war_if_valid = yes
							}
							hidden_effect = {
								if = {
									limit = { NOT = { exists = joined_faction.faction_war } }
									war_for_imprison_effect = {
										LIEGE = scope:puppet_or_actor
										REBEL = scope:recipient
									}
								}
							}
						}
						else = {
							war_for_imprison_effect = {
								LIEGE = scope:puppet_or_actor
								REBEL = scope:recipient
							}
						}
					}
					#If target is a baron or unlanded, have them escape.
					else = {
						if = { #To block them from being instantly rehired
							limit = {
								is_councillor_of = scope:puppet_or_actor
								can_be_fired_from_council_trigger = { COURT_OWNER = scope:puppet_or_actor }
							}
							set_variable = {
								name = escaped_imprisonment_from
								value = scope:puppet_or_actor
								years = 20
							}
						}
						imprison_court_chaplain_consequences_effect = yes
						if = {
							limit = {
								is_landed_or_landless_administrative_or_theocratic = yes
							}
							depose_effect = { DEPOSER = scope:puppet_or_actor }
						}
						else_if  = {
							limit = {
								is_hostage = yes
								warden = scope:puppet_or_actor
							}
							save_scope_as = imprison_failure
							save_scope_as = hostage
							warden = { save_scope_as = warden }
							home_court = { save_scope_as = home_court }
							scope:warden = { trigger_event = bp2_hostage_system.0400 }
							scope:home_court = { trigger_event = bp2_hostage_system.0401 }
							return_hostage = yes
						}
						else_if = {
							limit = {
								is_landed_or_landless_administrative_or_theocratic = no
								trigger_if = {
									limit = {
										is_councillor = yes
									}
									can_be_fired_from_council_trigger = { COURT_OWNER = scope:recipient.liege }
								}
								trigger_if = {
									limit = {
										exists = scope:puppet_or_actor.faith.religious_head
									}
									scope:puppet_or_actor.faith.religious_head != scope:recipient
								}
							}
							select_and_move_to_pool_effect = yes
						}
						add_opinion = {
							target = scope:puppet_or_actor
							modifier = attempted_imprisonment_opinion
						}
					}
				}

				if = {
					limit = {
						scope:puppet_or_actor = { has_realm_law_flag = vassal_refusal_is_treason }
					}
					scope:puppet_or_actor = {
						add_opinion = {
							target = scope:recipient
							modifier = treasonous_imprison_refusal
						}
					}
				}
			}

			#Remove them as guardian/ward for any courtier (feedback given in char_interaction.0230)
			scope:puppet_or_actor = {
				if = {
					limit = {
						any_courtier = {
							has_relation_guardian = scope:recipient
						}
					}
					every_courtier = {
						limit = {
							has_relation_guardian = scope:recipient
						}
						save_scope_as = ward
						scope:recipient = { remove_relation_ward = scope:ward }

						scope:ward = {
							if = {
								limit = {
									host != liege
								}
								return_to_court = yes
							}
							guardian_remove_opinion_effect = {
								GUARDIAN = scope:recipient
								WARD = this
								WARD_LIEGE = scope:puppet_or_actor
							}
						}

						add_to_list = ward_children
					}
				}
				if = {
					limit = {
						any_courtier = {
							has_relation_ward = scope:recipient
						}
					}
					random_courtier = {
						limit = {
							has_relation_ward = scope:recipient
						}
						save_scope_as = guardian
						scope:guardian = { remove_relation_ward = scope:recipient }

						scope:guardian = {
							if = {
								limit = {
									host != liege
								}
								return_to_court = yes
							}
							guardian_remove_opinion_effect = {

							}
							every_character_artifact = {
								limit = { NOT = { has_variable = artifact_prevent_transfer } }
								set_owner = {
									target = scope:puppet_or_actor
									history = {
										location = scope:recipient.location
										actor = scope:recipient
										recipient = scope:puppet_or_actor
										type = stolen
									}
								}
								scope:puppet_or_actor = {
									if = {
										limit = {
											NOT = { has_variable = stolen_artifact }
										}
										set_variable = {
											name = stolen_artifact
											value = 1
										}
									}
									else = {
										change_variable = {
											name = stolen_artifact
											add = 1
										}
									}
								}
							}
						}
						banish = yes
					}
					# 'Banish' Pool Character
					else_if = {
						limit = {
							is_playable_character = no
							is_pool_character = yes # They'll just leave for the pool
						}
						if = {
							limit = {
								gold > 0
							}
							pay_short_term_gold = { target = scope:puppet_or_actor gold = scope:recipient.gold }
						}
					}
					add_opinion = {
						modifier = banished_me
						target = scope:puppet_or_actor
					}
					release_from_prison = yes
				}
			}
			# Renounce Claims
			scope:renounce_claims = {
				scope:recipient = {
					add_opinion = {
						modifier = demanded_claim_renouncement
						target = scope:puppet_or_actor
					}
					release_from_prison = yes
					every_claim = {
						explicit = yes
						limit = {
							save_temporary_scope_as = temp_claim
							OR = {
								holder = scope:puppet_or_actor
								AND = {
									exists = holder
									NOT = { prev = { target_is_liege_or_above = scope:puppet_or_actor } }
									holder = { target_is_liege_or_above = scope:puppet_or_actor }
								}
								scope:puppet_or_actor = {
									any_held_title = {
										is_de_jure_liege_or_above_target = scope:temp_claim
									}
								}
							}
						}
						scope:recipient = { remove_claim = prev }
					}
				}
			}
			# Take Vows
			scope:take_vows = {
				send_child_to_clergy_effect = yes
				#Does Byzan go hay-wire with deposing vassals?
				if = {
					limit = {
						scope:puppet_or_actor = {
							is_ai = yes
							culture = { has_cultural_pillar = heritage_byzantine }
						}
					}
					debug_log = debug_ai_byzan_forced_vows_on_prison_release
					debug_log_scopes = yes
				}
				#Does the AI in general go haywire with deposing prisoners?
				else_if = {
					limit = {
						scope:puppet_or_actor = { is_ai = yes }
					}
					debug_log = debug_ai_forced_vows_on_prison_release
					debug_log_scopes = yes
				}
				scope:recipient = {
					release_from_prison = yes
				}
			}
		}
	}
