er_on_title_gain = {
	effect = {
		#Kenneth Haigh specific title gains
		if = {
			limit = {
				NOT = { exists = global_var:fort_haight_reclaimed }
				root = character:haight
				#house:house_haight.house_head = root
				scope:title = title:c_fort_haight
				game_start_date >= 8301
			}
			set_global_variable = fort_haight_reclaimed
			trigger_event = er_haight.0001 days = 1
		}
		
		if = {
			limit = {
				NOT = { exists = global_var:haight_expands_realm }
				exists = global_var:fort_haight_reclaimed
				root = character:haight
				scope:title = title:d_mistwood
				game_start_date >= 8301
			}
			set_global_variable = haight_expands_realm
			trigger_event = er_haight.0002 days = 1
		}
		
		if = {
			limit = {
				NOT = { exists = global_var:haight_expands_realm }
				exists = global_var:fort_haight_reclaimed
				root = character:haight
				realm_size >= 15
				#scope:title = title:d_mistwood
				game_start_date >= 8301
			}
			set_global_variable = haight_expands_realm
			trigger_event = er_haight.0002 days = 1
		}
		
		if = {
			limit = {
				NOT = { exists = global_var:haight_gains_stormveil }
				exists = global_var:haight_expands_realm
				root = character:haight
				scope:title = title:k_stormveil
				#has_title = title:c_stormveil_throne
				game_start_date >= 8301
			}
			set_global_variable = haight_gains_stormveil
			trigger_event = er_haight.0003 days = 1
		}
		
		if = {
			limit = {
				NOT = { exists = global_var:haight_gains_limgrave }
				#exists = global_var:haight_gains_stormveil
				root = character:haight
				scope:title = title:e_limgrave
				game_start_date >= 8301
			}
			set_global_variable = haight_gains_limgrave
			trigger_event = er_haight.0004 days = 1
		}
	}
}
