on_birth_child = {
    on_actions = {
        on_action_child_birth_god
		on_action_special_tarnished_produce_normal_ones
		on_action_marais_children_born_sick
		on_action_children_born_during_death_bound
    }
}

five_year_everyone_pulse = {
    on_actions = {
        on_action_ai_become_empyrean
		on_action_gods_or_dragons_immune_to_old_age
		on_action_become_immortal_becouse_death_bound
    }
}

on_action_child_birth_god = {
	effect = {
		if = {
			limit = {
				OR = {
					scope:real_father = {
						OR = {
							has_godhood_elden_ring = yes
							is_demigod_elden_ring = yes
						}
						
					}
					scope:mother = {
						OR = {
							has_godhood_elden_ring = yes
							is_demigod_elden_ring = yes
						}
					}
				}
			}
			if = { # Both parents are gods
				limit = {
					AND = {
						scope:real_father = {
							has_trait = god_elden_ring
						}
						scope:mother = {
							has_trait = god_elden_ring
						}
					}
				}
				add_trait = demigod_elden_ring_4
				add_trait = empyrean_elden_ring
				random_list = {
					10 = {
						add_trait = eternal_youth_elden_ring
					}
					90 = {
					
					}		
				}
			}
			else_if = { # one of the parents is a god
				limit = {
					OR = {
						scope:real_father = {
							has_trait = god_elden_ring
						}
						scope:mother = {
							has_trait = god_elden_ring
						}
					}
				}
				add_trait = demigod_elden_ring_4
			}
			else_if = { # the blood gets weaker 4->3
				limit = {
					OR = {
						scope:real_father = {
							has_trait = demigod_elden_ring_4
						}
						scope:mother = {
							has_trait = demigod_elden_ring_4
						}
					}
				}
				
				random_list = {
					1 = {
						add_trait = demigod_elden_ring_4
					}
					99 = {
						add_trait = demigod_elden_ring_3
					}
				}
			}
			else_if = { # the blood gets weaker 3->2
				limit = {
					OR = {
						scope:real_father = {
							has_trait = demigod_elden_ring_3
						}
						scope:mother = {
							has_trait = demigod_elden_ring_3
						}
					}
				}
				random_list = {
					1 = {
						add_trait = demigod_elden_ring_3
					}
					99 = {
						add_trait = demigod_elden_ring_2
					}		
				}
			}
			else_if = { # the blood gets weaker 2->1
				limit = {
					OR = {
						scope:real_father = {
							has_trait = demigod_elden_ring_2
						}
						scope:mother = {
							has_trait = demigod_elden_ring_2
						}
					}
				}
				random_list = {
					1 = {
						add_trait = demigod_elden_ring_2
					}
					99 = {
						add_trait = demigod_elden_ring_1
					}		
				}
			}
		}
		
	}
}

on_action_ai_become_empyrean = {
	effect = {
		if = { # emperor or more that is VERY religious
			limit = {
				AND = {
					is_independent_ruler = yes
					NOR = {
						has_trait = empyrean_elden_ring
						#has_trait = demigod_elden_ring
						has_trait = god_elden_ring
					}
					highest_held_title_tier >= tier_empire
					piety_level >= 5
				}
			}
			random_list = {
				5 = {
					add_trait = empyrean_elden_ring
				}
				95 = {
				
				}		
			}
		}
				
	}
}

on_action_gods_or_dragons_immune_to_old_age = {
	effect = {
		if = {
			limit = {
				OR = {
					has_godhood_elden_ring = yes
					is_demigod_elden_ring = yes
						
					has_trait = ancient_dragon_elden_ring
					has_trait = lesser_dragon_elden_ring
				}
				
			}
			remove_trait = disfigured
			remove_trait = incapable
			remove_trait = cancer
		}
				
	}
}

on_action_special_tarnished_produce_normal_ones = {
	effect = {
		if = {
			limit = {
				OR = {
					scope:real_father = {
						is_tarnished = yes
					}
					scope:mother = {
						is_tarnished = yes
					}
				}
			}
			add_trait = tarnished			
		}
	}
}

on_action_marais_children_born_sick = {
	effect = {
		if = {
			limit = {
				is_male = yes
				dynasty = dynasty:dynn_marais
			}
			add_trait = sickly			
		}
	}
}

on_action_children_born_during_death_bound = {
	effect = {
		if = {
			limit = {
				global_var:rune_of_death_bound = yes
			}
			add_character_modifier = gift_from_the_eternal
		}
	}
}

on_action_become_immortal_becouse_death_bound = {
	effect = {
		if = {
			limit = {
				global_var:rune_of_death_bound = yes
				NOT = {
					has_character_modifier = gift_from_the_eternal
				}
			}
			add_character_modifier = gift_from_the_eternal		
		}
	}
}
