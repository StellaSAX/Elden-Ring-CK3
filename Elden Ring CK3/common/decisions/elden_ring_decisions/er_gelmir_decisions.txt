expand_gelmir = {
	ai_check_interval = 72

	picture = { reference = "gfx/interface/illustrations/decisions/decision_expand_gelmir.dds" }

	decision_group_type = major

	is_shown = {
		title:d_wyndham = {
			any_this_title_or_de_jure_above = {
				NOT = {
					this = title:k_gelmir
				}
			}
		}
		title:d_coffin_lake = {
			any_this_title_or_de_jure_above = {
				NOT = {
					this = title:k_gelmir
				}
			}
		}
		NOT = { government_has_flag = government_is_tribal }
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:gelmir_expansion
			}
		}
		is_independent_ruler = yes
		highest_held_title_tier >= tier_kingdom
		has_primary_title = title:k_gelmir
		
		OR = {
			is_rykard = yes
			OR = {
				culture = culture:cul_gelmir
				culture = { any_parent_culture_or_above = { this = culture:cul_gelmir } }
			}
		}
	}
	
	is_valid = {
		#prestige_level >= 2
		completely_controls_region = region_gelmir
		completely_controls = title:d_wyndham
		completely_controls = title:d_coffin_lake
	}
	
	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_adult = yes
		is_at_war = no
	}
	
	cost = {
		prestige = {
			if = {
				limit = {
					legitimacy_level = 1
				}
				value = 1000
			}
			if = {
				limit = {
					legitimacy_level = 2
				}
				value = 750
			}
			if = {
				limit = {
					legitimacy_level = 3
				}
				value = 500
			}
			if = {
				limit = {
					legitimacy_level = 4
				}
				value = 350
			}
			if = {
				limit = {
					legitimacy_level = 5
				}
				value = 250
			}
		}
	}

	effect = {
		trigger_event = er_gelmir.0001
		title:d_wyndham = { set_de_jure_liege_title = title:k_gelmir }
		title:d_coffin_lake = { set_de_jure_liege_title = title:k_gelmir }
		add_legitimacy = medium_legitimacy_gain
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:gelmir_expansion
			}
		}
	}
	
	ai_potential = { always = yes }
	ai_will_do = { base = 25 }
}
