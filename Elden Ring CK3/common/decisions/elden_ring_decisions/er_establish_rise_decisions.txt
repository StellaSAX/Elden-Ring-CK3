@establish_rise_base_gold_cost = 750

er_establish_rise = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/pam_decision_act_of_devotion.dds"
	}
	decision_group_type = major

	# ai_goal = yes

	is_shown = {
		is_landed = yes
		culture = {
			OR = {
				has_cultural_tradition = tradition_carian_royal
				has_cultural_tradition = tradition_magic_sorcery_carian
			}
		}
		NOT = { has_variable = er_rise_owned }
	}

	widget = {
		controller = create_holy_order
		title_valid = {
			trigger_if = {
				limit = { exists = this }
				title_province = {
					NOR = {
						terrain = lava_lake
						terrain = narrow_path
						terrain = scarlet_swamp
					}
					has_special_building = no
				}
			}
			trigger_else = {
				always = no
			}
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
	}

	is_valid = {
		save_temporary_scope_as = sorcerer
		custom_tooltip = {
			text = must_be_sorcerer_expert_or_above
			OR = {
				has_trait_xp = {
					trait = sorcery_carian
					value >= 80
				}
				has_trait_xp = {
					trait = sorcery_glintstone
					value >= 80
				}
			}
		}
		OR = {
			prestige_level >= 4
			custom_tooltip = {
				text = must_be_famous_sorcerer
				OR = {
					this = character:rennala
					this = character:ranni
					this = character:seluvis
				}
			}
		}
	}
	cost = {
		gold = {
			value = 0
			if = {
				limit = { has_treasury = no }
				add = @establish_rise_base_gold_cost
				if = { # Double the cost if on poison swamps or other water-related provinces
					limit = {
						scope:barony ?= {
							title_province = {
								OR = {
									terrain = bog_poison
									terrain = wetlands
								}
							}
						}
					}
					multiply = 2
				}
			}
		}
		treasury = {
			value = 0
			if = {
				limit = { has_treasury = yes }
				add = @establish_rise_base_gold_cost
				if = { # Double the cost if on poison swamps or other water-related provinces
					limit = {
						scope:barony ?= {
							title_province = {
								OR = {
									terrain = bog_poison
									terrain = wetlands
								}
							}
						}
					}
					multiply = 2
				}
			}
		}
		piety = {
			value = 1
		}
	}

	effect = {
		save_scope_as = founder
		if = {
			limit = { exists = scope:barony }
			scope:barony = {
				set_variable = {
					name = er_rise_founder
					value = scope:founder
				}
				title_province = {
					add_special_building = generic_rise
				}
			}
			set_variable = {
				name = er_rise_owned
				value = scope:barony
			}
			trigger_event = er_establish_rise.0001
			every_player = {
				limit = {
					NOT = { this = scope:founder }
					OR = {
						is_close_family_of = scope:founder
						OR = {
							has_relation_best_friend = scope:founder
							has_relation_friend = scope:founder
							has_relation_nemesis = scope:founder
							has_secret_relation_nemesis = scope:founder
							has_relation_rival = scope:founder
							has_relation_potential_rival = scope:founder
						}
					}
				}
				save_scope_as = receiver
				send_interface_message = {
					type = send_interface_message
					title = er_establish_rise.relation_toast.title
					custom_tooltip = er_establish_rise.relation_toast.tt

					left_icon = scope:founder
					right_icon = scope:barony
				}
			}
		}
	}
	
	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
