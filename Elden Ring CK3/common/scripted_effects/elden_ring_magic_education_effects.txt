resolve_magic_education_success_reward_effect = {
	involved_activity ?= {
		#Perfect studies - 75-100
		if = {
			limit = { var:activity_special_type_progression >= 75 }
			scope:host = {
				add_character_flag = perfect_studies_reward
				
				if = { #doesn't have trait
					limit = { has_magic_education_trait_trigger = no }
					add_character_flag = get_new_magic
				}
				else_if = { #has trait and intelligent
					limit = { 
						has_magic_education_trait_trigger = yes 
						probably_intelligent_trigger = yes 
					}
					random = {
						chance = 75
						add_character_flag = get_new_magic
					}
					add_character_flag = get_magic_xp
				}
				else_if = { #has trait so gain xp
					limit = { 
						has_magic_education_trait_trigger = yes 
					}
					add_character_flag = get_magic_xp
				}
				
			}
		}
		#High-end studies - 50-75
		else_if = {
			limit = { 
				var:activity_special_type_progression < 75
				var:activity_special_type_progression >= 50
			} 
			scope:host = {
				add_character_flag = high_studies_reward
				if = { #doesn't have trait
					limit = { has_magic_education_trait_trigger = no }
					add_character_flag = get_new_magic
				}
				else_if = { #has trait and intelligent
					limit = { 
						has_magic_education_trait_trigger = yes 
						probably_intelligent_trigger = yes 
					}
					random = {
						chance = 50
						add_character_flag = get_new_magic
					}
					add_character_flag = get_magic_xp
				}
				else_if = { #has trait so gain xp
					limit = { 
						has_magic_education_trait_trigger = yes 
					}
					add_character_flag = get_magic_xp
				}
			}
		}
		#OK studies - 25-50
		else_if = {
			limit = {
				var:activity_special_type_progression < 50
				var:activity_special_type_progression >= 25
			}
			scope:host = {
				add_character_flag = mid_studies_reward
				if = { #doesn't have trait
					limit = { has_magic_education_trait_trigger = no }
					add_character_flag = get_new_magic
				}
				else_if = { #has trait and intelligent
					limit = { 
						has_magic_education_trait_trigger = yes 
						probably_intelligent_trigger = yes 
					}
					random = {
						chance = 25
						add_character_flag = get_new_magic
					}
					add_character_flag = get_magic_xp
				}
				else_if = { #has trait so gain xp
					limit = { 
						has_magic_education_trait_trigger = yes 
					}
					add_character_flag = get_magic_xp
				}
			}
		}
		#Lousy studies - 0-25
		else = {
			scope:host = {
				add_character_flag = low_studies_reward
				if = { #doesn't have trait
					limit = { has_magic_education_trait_trigger = no }
					add_character_flag = get_new_magic
				}
				else_if = { #has trait and intelligent
					limit = { 
						has_magic_education_trait_trigger = yes 
						probably_intelligent_trigger = yes 
					}
					random = {
						chance = 10
						add_character_flag = get_new_magic
					}
					add_character_flag = get_magic_xp
				}
				else_if = { #has trait so gain xp
					limit = { 
						has_magic_education_trait_trigger = yes 
					}
					add_character_flag = get_magic_xp
				}
			}
		}
	}
}

magic_education_completed_log_entry_effect = {
	involved_activity = {
		add_activity_log_entry = {
			key = adult_education_studies_completed_log
			tags = { completed }
			score = 100
			show_in_conclusion = yes
			character = root
			#Effects
			root = {
				disburse_magic_education_reward_effect = yes
			}
		}
	}
}

disburse_magic_education_reward_effect = {
	switch = {
		trigger = has_character_flag
		get_new_magic = {
			get_new_magic_effect = yes
			remove_character_flag = get_new_magic
		}
		get_magic_xp = {
			get_magic_xp_effect = yes
			remove_character_flag = get_magic_xp
		}
	}

}

get_new_magic_effect = {
	if = {
		limit = {
			scope:activity = {
				has_activity_option = {
					category = magic_education_option_specialization
					option = learn_local_magic
				}
			}
		}
		get_local_magic_effect = yes
	}
	else_if = {
		limit = {
			scope:activity = {
				has_activity_option = {
					category = magic_education_option_specialization
					option = learn_affinity_magic
				}
			}
		}
		get_affinity_magic_effect = yes
	}
	else_if = {
		limit = {
			scope:activity = {
				has_activity_option = {
					category = magic_education_option_specialization
					option = learn_cultural_magic
				}
			}
		}
		get_cultural_magic_effect = yes
	}
	else_if = {
		limit = {
			scope:activity = {
				has_activity_option = {
					category = magic_education_option_specialization
					option = learn_religious_magic
				}
			}
		}
		get_religious_magic_effect = yes
	}
	else = {
		learn_random_magic_effect = yes
	}
}

get_local_magic_effect = {
	if = {
		limit = {
			location = {
				has_building_with_flag = golden_order_magic
			}
		}
		random_list = {
			34 = {
				modifier = {
					add = 30
					has_trait = erdtree_affinity
				}
				add_trait = incantation_erdtree 
			}
			33 = { 
				modifier = {
					add = 30
					has_trait = order_affinity
				}
				add_trait = incantation_order 
			}
			33 = { add_trait = incantation_two_fingers }
		}
	}
	if = {
		limit = {
			location = {
				has_building_with_flag = cosmos_magic
			}
		}
		random_list = {
			34 = { add_trait = sorcery_glintstone }
			33 = { add_trait = sorcery_carian }
			33 = { add_trait = sorcery_gravity }
		}
	}
	if = {
		limit = {
			location = {
				has_building_with_flag = dragon_magic
			}
		}
		random_list = {
			50 = { add_trait = incantation_dragon_communion }
			50 = { add_trait = incantation_dragon_cult }
		}
	}
	
}
