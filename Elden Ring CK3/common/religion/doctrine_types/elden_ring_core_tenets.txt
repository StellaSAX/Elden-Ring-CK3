		else = {
			value = faith_tenet_cost_high
		}

		# Multiplier for keeping same tenet

			limit = { has_doctrine = tenet_primordial_curse }

		}
	}

		blessed_by_the_crucible = yes #for giving opinion bonuses to traits that could be considered "deformities"
