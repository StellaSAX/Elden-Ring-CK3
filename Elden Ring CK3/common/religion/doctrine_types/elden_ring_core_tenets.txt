tenet_primordial_curse = {
	icon = tenet_primordial_curse
	piety_cost = {
		if = {
			limit = { religion_tag = crucible_religion_elden_ring }
			value = faith_tenet_cost_low
		}
		else = {
			value = faith_tenet_cost_high
		}

		# Multiplier for keeping same tenet
		if = {
			limit = { has_doctrine = tenet_primordial_curse }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	is_shown = {
		# OR = {
			# religion_tag = crucible_religion_elden_ring
		# }
	}

	can_pick = {
		custom_description = {
			text = incompatible_tenet_golden_order_desc
			NOT = { doctrine:tenet_golden_order = { is_in_list = selected_doctrines } }	
		}
	}

	parameters = {
		
	}
	character_modifier = {
		health = 0.5		#Representing the vitality that the crucible grants to one
		
	}
	
	traits = {
		virtues = { crucible_affinity crucible_knight }
	}
}
